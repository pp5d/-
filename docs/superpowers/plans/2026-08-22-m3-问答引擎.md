# M3 问答引擎 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现问答引擎核心——知识向量化（本地 BGE-M3）、向量检索（余弦相似度）、DeepSeek 生成带来源的回答、前端问答对话页。

**Architecture:** 在 M2 知识库基础上扩展。知识发布时自动分块并向量化存入 `chunks` 表（JSON 列存 1024 维向量，不用 pgvector）；提问时向量化问题 → Python 端余弦相似度检索 top-k 片段 → 组装 prompt → DeepSeek 生成回答并附来源。未命中（相似度低于阈值）返回"转人工"提示（工单闭环属 M4）。

**Tech Stack:** FastAPI / SQLAlchemy 2.0 / sentence-transformers(BGE-M3) / httpx(DeepSeek) / Vue3 / Element Plus

## Global Constraints

- **开发环境（务必遵守）：**
  - 后端 venv：`D:\Vibing Code Project\aiqa\backend\.venv\Scripts\python.exe`（简称 `<venvPy>`）；跑 pytest 用它（系统 Python 不可写）。
  - PostgreSQL 16：`localhost:5432`，角色 `aiqa`/密码 `aiqa_dev_password`；测试库 `aiqa_test` 由 conftest 自动创建。
  - **向量模型 BGE-M3**：通过 HuggingFace 镜像 `HF_ENDPOINT=https://hf-mirror.com` 下载（本机直连 HuggingFace 不通）。`services/embedding.py` 中 `os.environ.setdefault("HF_ENDPOINT", "https://hf-mirror.com")` 必须在 import sentence_transformers 之前设置。
  - **模型名**：`BAAI/bge-m3`（1024 维）。若下载失败可降级 `BAAI/bge-small-zh-v1.5`（512 维，约 100MB）——降级时需同步改 `embedding.py` 的模型名，其余代码不变。
  - **DeepSeek API**：`settings.DEEPSEEK_API_KEY`（用户已有）；base_url `https://api.deepseek.com`，model `deepseek-chat`；用 httpx 调用（不引入 openai SDK）。
  - 沙箱禁止 esbuild spawn，前端任务用 `@vue/compiler-sfc` 静态编译验证。
- **向量存储**：`chunks.embedding` 用 JSON 列存浮点数组（不用 pgvector）；向量已 normalize，余弦相似度=点积。
- **检索范围**：公网模式只检索 `published` 且 `publish_to_public=True` 的知识；内网检索全部 `published`。
- **未命中阈值**：`SIM_THRESHOLD = 0.35`（归一化余弦相似度）。低于阈值返回 `hit=False`，提示转人工（工单落地在 M4）。
- **知识发布时向量化**：`review` 接口 `approve=True` 时，调用 `services.indexing.reindex_knowledge(item, db)` 重建该知识的 chunks（先删旧再建）。

- [ ] **Step 1: 创建 services/indexing.py**

```python
from sqlalchemy import delete, select

from app.models.chunk import Chunk
from app.models.knowledge import KnowledgeItem
from app.services.embedding import embed_texts


def chunk_text(text: str, max_len: int = 500) -> list[str]:
    paragraphs = [p.strip() for p in (text or "").split("\n") if p.strip()]
    chunks: list[str] = []
    cur = ""
    for p in paragraphs:
        if len(cur) + len(p) > max_len and cur:
            chunks.append(cur)
            cur = p
        else:
            cur = (cur + "\n" + p).strip()
    if cur:
        chunks.append(cur)
    if not chunks:
        chunks = [(text or "")[:max_len]]
    return chunks


def reindex_knowledge(item: KnowledgeItem, db) -> None:
    db.execute(delete(Chunk).where(Chunk.knowledge_id == item.id))
    texts = chunk_text(item.body) + list(item.aliases or [])
    if not texts:
        return
    vectors = embed_texts(texts)
    for i, (t, v) in enumerate(zip(texts, vectors)):
        db.add(Chunk(knowledge_id=item.id, chunk_index=i, content=t, embedding=v))
```

（注意：`reindex_knowledge` **不 commit**，由调用方（review 函数）统一 commit，保证状态与向量原子提交。）

```python
from sqlalchemy import select

from app.config import settings
from app.models.chunk import Chunk
from app.models.knowledge import KnowledgeItem
from app.services.embedding import embed_texts

SIM_THRESHOLD = 0.35


def _cosine(a: list[float], b: list[float]) -> float:
    return sum(x * y for x, y in zip(a, b))


def search(question: str, db, top_k: int = 5) -> list[tuple[float, Chunk, KnowledgeItem]]:
    q = embed_texts([question])[0]
    stmt = (
        select(Chunk, KnowledgeItem)
        .join(KnowledgeItem, KnowledgeItem.id == Chunk.knowledge_id)
        .where(KnowledgeItem.status == "published")
    )
    if settings.APP_MODE == "public":
        stmt = stmt.where(KnowledgeItem.publish_to_public == True)  # noqa: E712
    rows = db.execute(stmt).all()
    scored = [(_cosine(q, c.embedding), c, item) for c, item in rows]
    scored.sort(key=lambda x: -x[0])
    return scored[:top_k]
```

```python
import httpx

from app.config import settings


def build_prompt(question: str, contexts: list[str]) -> str:
    parts = ["以下是注塑机上位机领域的知识片段："]
    for i, c in enumerate(contexts, 1):
        parts.append(f"[{i}] {c}")
    parts.append(
        f"\n请仅基于以上知识片段回答用户问题，若片段中没有答案请明确说"知识库暂无相关内容"。\n用户问题：{question}"
    )
    return "\n".join(parts)


def generate(question: str, contexts: list[str]) -> str:
    prompt = build_prompt(question, contexts)
    r = httpx.post(
        "https://api.deepseek.com/chat/completions",
        headers={"Authorization": f"Bearer {settings.DEEPSEEK_API_KEY}"},
        json={
            "model": "deepseek-chat",
            "messages": [
                {"role": "system", "content": "你是注塑机上位机技术助手，只基于提供的知识片段回答，不编造。"},
                {"role": "user", "content": prompt},
            ],
            "temperature": 0.3,
        },
        timeout=60,
    )
    r.raise_for_status()
    return r.json()["choices"][0]["message"]["content"]
```

```python
from fastapi import APIRouter, Depends
from sqlalchemy.orm import Session

from app.core.deps import get_current_user
from app.db import get_db
from app.models.user import User
from app.schemas.qa import QaIn, QaOut, QaSource
from app.services.llm import generate
from app.services.retrieval import SIM_THRESHOLD, search

router = APIRouter(prefix="/api/qa", tags=["qa"])


@router.post("", response_model=QaOut)
def ask(body: QaIn, user: User = Depends(get_current_user), db: Session = Depends(get_db)):
    results = search(body.question, db)
    if not results or results[0][0] < SIM_THRESHOLD:
        return QaOut(answer="知识库中暂未找到相关答案，问题已记录，工程师会尽快补充。", sources=[], hit=False)
    contexts = [c.content for _, c, _ in results]
    answer = generate(body.question, contexts)
    seen = set()
    sources = []
    for _, _, item in results:
        if item.id not in seen:
            seen.add(item.id)
            sources.append(QaSource(id=item.id, title=item.title))
    return QaOut(answer=answer, sources=sources, hit=True)
```

<script setup>
import { nextTick, onMounted, ref } from 'vue'
import http from '../../api/http'

const messages = ref([{ role: 'assistant', text: '你好！我是注塑机上位机智能助手，请描述你遇到的问题。' }])
const question = ref('')
const loading = ref(false)
const msgBox = ref(null)

async function send() {
  const q = question.value.trim()
  if (!q || loading.value) return
  messages.value.push({ role: 'user', text: q })
  question.value = ''
  loading.value = true
  try {
    const r = await http.post('/qa', { question: q })
    messages.value.push({ role: 'assistant', text: r.answer, sources: r.sources, hit: r.hit })
  } catch (e) {
    messages.value.push({ role: 'assistant', text: '请求失败，请稍后重试。' })
  } finally {
    loading.value = false
    nextTick(() => { msgBox.value && (msgBox.value.scrollTop = msgBox.value.scrollHeight) })
  }
}

onMounted(() => {})
